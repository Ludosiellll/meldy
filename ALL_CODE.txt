async def event_stream(
    file: UploadFile,
    summary_detail: str = "medium",
    flashcard_count: int = 6,
    quiz_count: int = 5,
    quiz_difficulty: str = "mixed",
    quiz_type: str = "multiple",
) -> AsyncGenerator[str, None]:
    """SSE stream of agent progress."""
    try:
        contents = await file.read()
        text = extract_text(contents, file.filename or "file.txt")

        if not text.strip():
            yield f"data: {json.dumps({'type': 'error', 'message': 'Не удалось извлечь текст из файла'})}\n\n"
            return

        yield f"data: {json.dumps({'type': 'text_extracted', 'length': len(text), 'text': text})}\n\n"

        coordinator = CoordinatorAgent(api_key=GROQ_API_KEY)

        async for event in coordinator.process(
            text,
            summary_detail=summary_detail,
            flashcard_count=flashcard_count,
            quiz_count=quiz_count,
            quiz_difficulty=quiz_difficulty,
            quiz_type=quiz_type,
        ):
            yield f"data: {json.dumps(event)}\n\n"

        yield f"data: {json.dumps({'type': 'done'})}\n\n"

    except Exception as e:
        yield f"data: {json.dumps({'type': 'error', 'message': str(e)})}\n\n"


@app.post("/api/analyze")
async def analyze(
    file: UploadFile = File(...),
    summary_detail: str = Form("medium"),
    flashcard_count: int = Form(6),
    quiz_count: int = Form(5),
    quiz_difficulty: str = Form("mixed"),
    quiz_type: str = Form("multiple"),
):
    if not GROQ_API_KEY:
        raise HTTPException(
            status_code=500,
            detail="GROQ_API_KEY не настроен на сервере",
        )

    allowed_extensions = (".pdf", ".txt", ".docx", ".md", ".doc")
    ext = os.path.splitext(file.filename or "")[1].lower()
    if ext not in allowed_extensions:
        raise HTTPException(
            status_code=400,
            detail=f"Неподдерживаемый формат. Поддерживаются: {', '.join(allowed_extensions)}",
        )

    return StreamingResponse(
        event_stream(file, summary_detail, flashcard_count, quiz_count, quiz_difficulty, quiz_type),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )

import io
from typing import Optional


def extract_text(contents: bytes, filename: str) -> str:
    ext = filename.lower().rsplit(".", 1)[-1] if "." in filename else "txt"

    if ext == "pdf":
        return _extract_pdf(contents)
    elif ext == "docx":
        return _extract_docx(contents)
    elif ext in ("txt", "md", "doc"):
        return _extract_text(contents)
    else:
        return _extract_text(contents)


def _extract_pdf(contents: bytes) -> str:
    try:
        from PyPDF2 import PdfReader
        reader = PdfReader(io.BytesIO(contents))
        text_parts = []
        for page in reader.pages:
            page_text = page.extract_text()
            if page_text:
                text_parts.append(page_text)
        return "\n\n".join(text_parts)
    except Exception as e:
        raise ValueError(f"Ошибка чтения PDF: {e}")


def _extract_docx(contents: bytes) -> str:
    try:
        from docx import Document
        doc = Document(io.BytesIO(contents))
        text_parts = []
        for para in doc.paragraphs:
            if para.text.strip():
                text_parts.append(para.text)
        return "\n".join(text_parts)
    except Exception as e:
        raise ValueError(f"Ошибка чтения DOCX: {e}")


def _extract_text(contents: bytes) -> str:
    for encoding in ("utf-8", "cp1251", "latin-1"):
        try:
            return contents.decode(encoding)
        except UnicodeDecodeError:
            continue
    return contents.decode("utf-8", errors="replace")

import os
import random
import asyncio
from groq import AsyncGroq
import google.generativeai as genai
from google.generativeai.types import SafetySettingDict, HarmCategory, HarmBlockThreshold

# ... (промпты без изменений)

class BaseAgent:
    def __init__(self, api_key: str, model: str = "llama-3.3-70b-versatile"):
        self.groq_api_key = api_key
        self.model = model
        
        # Читаем ключи
        raw_keys = os.getenv("GEMINI_API_KEY", "")
        self.gemini_keys = [k.strip() for k in raw_keys.split(",") if k.strip()]
        
        if self.groq_api_key:
            self.groq_client = AsyncGroq(api_key=self.groq_api_key)
        else:
            self.groq_client = None

        self.safety_settings: list[SafetySettingDict] = [
            {"category": HarmCategory.HARM_CATEGORY_HARASSMENT, "threshold": HarmBlockThreshold.BLOCK_NONE},
            {"category": HarmCategory.HARM_CATEGORY_HATE_SPEECH, "threshold": HarmBlockThreshold.BLOCK_NONE},
            {"category": HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT, "threshold": HarmBlockThreshold.BLOCK_NONE},
            {"category": HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT, "threshold": HarmBlockThreshold.BLOCK_NONE},
        ]

    async def run(self, system_prompt: str, user_text: str) -> str:
        max_chars = 300000 if self.gemini_keys else 80000
        truncated = user_text[:max_chars]
        if len(user_text) > max_chars:
            truncated += f"\n\n[Текст сокращён с {len(user_text)} до {max_chars} символов]"

        gemini_errors = []

        if self.gemini_keys:
            # Пробуем ключи по очереди
            shuffled_keys = list(self.gemini_keys)
            random.shuffle(shuffled_keys)
            
            for key in shuffled_keys:
                try:
                    # Настраиваем библиотеку под конкретный ключ
                    genai.configure(api_key=key)
                    
                    GEMINI_MODELS = [
                        "gemini-3.1-flash-lite",
                        "gemini-2.5-flash-preview-05-20",
                        "gemini-2.0-flash",
                        "gemini-1.5-flash",
                    ]
                    
                    model_loaded = None
                    model_error = None
                    for model_name in GEMINI_MODELS:
                        try:
                            model_loaded = genai.GenerativeModel(model_name)
                            break
                        except Exception as e:
                            model_error = str(e)
                            continue
                    
                    if not model_loaded:
                        gemini_errors.append(f"Не удалось загрузить ни одну Gemini модель: {model_error}")
                        continue  # переходим к следующему ключу
                    
                    full_prompt = f"{system_prompt}\n\nЗАПРОС:\n{truncated}"
                    
                    # Делаем запрос
                    response = await model_loaded.generate_content_async(
                        full_prompt,
                        generation_config=genai.types.GenerationConfig(
                            temperature=0.3,
                            max_output_tokens=8192,
                        ),
                        safety_settings=self.safety_settings
                    )
                    
                    if response and response.text:
                        return response.text
                    
                    gemini_errors.append(f"Key {key[:6]}... returned empty text")
                except Exception as e:
                    err_msg = str(e)
                    gemini_errors.append(f"Key {key[:6]}... error: {err_msg[:100]}")
                    continue

        # Если дошли сюда - Gemini не сработал. Пробуем Groq.
        if self.groq_client:
            try:
                response = await self.groq_client.chat.completions.create(
                    model=self.model,
                    messages=[
                        {"role": "system", "content": system_prompt},
                        {"role": "user", "content": truncated},
                    ],
                    temperature=0.3,
                    max_tokens=8192,
                )
                return response.choices[0].message.content or ""
            except Exception as e:
                groq_err = str(e)
                if "429" in groq_err:
                    debug_info = "\n".join(gemini_errors) if gemini_errors else "Ключи Gemini не найдены"
                    return f"Ошибка: Лимиты ИИ исчерпаны.\n\nGemini ошибки:\n{debug_info}\n\nGroq ошибка: 429."
                return f"Ошибка ИИ (Groq): {groq_err}"
        
        return "Ошибка: API ключи не настроены."

import asyncio
import json
from typing import AsyncGenerator, Dict, Any, Optional

from .base import BaseAgent, build_summarizer_prompt, build_flashcards_prompt, build_quiz_prompt


class CoordinatorAgent:
    def __init__(self, api_key: str):
        self.base = BaseAgent(api_key)

    async def process(
        self,
        text: str,
        summary_detail: str = "medium",
        flashcard_count: int = 6,
        quiz_count: int = 5,
        quiz_difficulty: str = "mixed",
        quiz_type: str = "multiple",
    ) -> AsyncGenerator[Dict[str, Any], None]:
        results: Dict[str, Any] = {}

        prompts = {
            "summarizer": build_summarizer_prompt(summary_detail),
            "flashcards": build_flashcards_prompt(flashcard_count),
            "quiz": build_quiz_prompt(quiz_count, quiz_difficulty, quiz_type),
        }

        async def run_agent(name: str, prompt: str) -> tuple:
            try:
                raw = await self.base.run(prompt, text)
                return (name, self._parse_result(name, raw))
            except Exception as e:
                return (name, {"error": str(e)})

        agent_labels = {
            "summarizer": "📝 Конспект",
            "flashcards": "🃏 Карточки",
            "quiz": "❓ Тест",
        }

        # Запускаем агентов ПОСЛЕДОВАТЕЛЬНО для стабильности (особенно для Gemini Free)
        done_count = 0
        for name in agent_names:
            yield {
                "type": "agent_start",
                "agent": name,
                "label": agent_labels[name],
            }
            
            # Выполняем запрос
            agent_name, result = await run_agent(name, prompts[name])
            results[agent_name] = result
            
            done_count += 1
            yield {
                "type": "agent_done",
                "agent": agent_name,
                "label": agent_labels[agent_name],
                "progress": done_count,
                "total": 3,
            }
            
            # Небольшая пауза между запросами для обхода лимитов
            if done_count < 3:
                await asyncio.sleep(1)

        yield {
            "type": "results",
            "summarizer": results.get("summarizer", {"error": "Нет результата"}),
            "flashcards": results.get("flashcards", {"error": "Нет результата"}),
            "quiz": results.get("quiz", {"error": "Нет результата"}),
        }

    def _parse_result(self, name: str, raw: str) -> Any:
        if name in ("flashcards", "quiz"):
            try:
                json_str = raw.strip()
                # Try to extract from markdown blocks
                if "```json" in json_str:
                    json_str = json_str.split("```json")[1].split("```")[0]
                elif "```" in json_str:
                    json_str = json_str.split("```")[1].split("```")[0]
                
                # If still no luck, try to find the first '{' or '[' and last '}' or ']'
                if not (json_str.startswith('{') or json_str.startswith('[')):
                    start_idx = -1
                    for char in ('{', '['):
                        idx = json_str.find(char)
                        if idx != -1 and (start_idx == -1 or idx < start_idx):
                            start_idx = idx
                    
                    if start_idx != -1:
                        end_char = '}' if json_str[start_idx] == '{' else ']'
                        end_idx = json_str.rfind(end_char)
                        if end_idx != -1:
                            json_str = json_str[start_idx:end_idx+1]

                parsed = json.loads(json_str.strip())
                
                # Normalize results: ensure {"quiz": [...]} or {"flashcards": [...]}
                if isinstance(parsed, list):
                    return {name: parsed}
                if isinstance(parsed, dict):
                    if name not in parsed and any(isinstance(v, list) for v in parsed.values()):
                        # If the key is different but contains a list, rename the first list found
                        for k, v in parsed.items():
                            if isinstance(v, list):
                                return {name: v}
                    return parsed
                
                return {"error": "Invalid format", "raw": raw}
            except Exception as e:
                return {"error": f"Parse error: {str(e)}", "raw": raw}
        return {"content": raw}

export const DEMO_TEXT = `Нейронные сети и глубокое обучение

Нейронная сеть — это математическая модель, вдохновлённая структурой биологического мозга. Она состоит из слоёв искусственных нейронов, каждый из которых получает входные сигналы, умножает их на веса, суммирует и пропускает через функцию активации. Результат передаётся следующему слою, образуя цепочку преобразований от входа к выходу.

Архитектура нейронной сети

Типичная полносвязная нейронная сеть включает три типа слоёв. Входной слой принимает исходные данные — например, пиксели изображения или признаки объекта. Скрытые слои выполняют последовательные нелинейные преобразования, извлекая всё более абстрактные представления. Выходной слой выдаёт результат: класс объекта, числовое предсказание или распределение вероятностей.

Количество скрытых слоёв определяет глубину сети. Сети с одним-двумя скрытыми слоями называются неглубокими, а сети с десятками слоёв — глубокими. Именно глубокие сети способны автоматически выделять сложные иерархические признаки, что делает их столь эффективными в задачах распознавания образов.

Функции активации

Функция активации определяет выход нейрона и вводит нелинейность в модель. Без нелинейности многослойная сеть сводилась бы к одному линейному преобразованию. Наиболее распространённые функции:

Сигмоида преобразует выход в интервал от нуля до единицы, что удобно для вероятностной интерпретации. Однако она страдает от проблемы затухающего градиента: при больших по модулю аргументах производная стремится к нулю, замедляя обучение глубоких сетей.

ReLU (Rectified Linear Unit) возвращает максимум из нуля и входного значения. Она вычислительно проста и устраняет затухание градиента для положительных значений, но может приводить к мёртвым нейронам, которые перестают активироваться.

GELU и Swish — современные функции, сочетающие преимущества ReLU с плавным переходом через ноль. Они показывают лучшие результаты в трансформерных архитектурах.

Обучение нейронной сети

Обучение сети заключается в подборе весов, минимизирующих функцию потерь — меру расхождения между предсказаниями и истинными значениями. Процесс состоит из повторяющихся шагов.

Прямой проход: данные подаются на вход, проходят через все слои, и сеть выдаёт предсказание. Вычисляется функция потерь.

Обратное распространение ошибки: с помощью правила дифференцирования сложной функции градиент потерь по каждому весу распространяется от выходного слоя к входному. Этот градиент показывает, как изменение веса повлияет на ошибку.

Градиентный спуск: веса обновляются в направлении, противоположном градиенту, с шагом, определяемым скоростью обучения. Маленькая скорость обеспечивает стабильность, но замедляет сходимость; большая ускоряет обучение, но может вызвать осцилляции.

Оптимизаторы Adam и RMSProp адаптируют скорость обучения для каждого параметра индивидуально, учитывая историю градиентов. Это значительно ускоряет сходимость по сравнению с ванильным градиентным спуском.

Регуляризация и борьба с переобучением

Переобучение — главная проблема глубоких сетей. Модель с большим числом параметров может запомнить обучающую выборку вместо выявления общих закономерностей. Для борьбы применяют несколько методов.

Dropout случайно отключает часть нейронов во время обучения, заставляя сеть распределять информацию между разными путями. На этапе вывода все нейроны активны, но их выходы масштабируются.

L2-регуляризация добавляет к функции потерь штраф за большие значения весов, предпочтая простые модели. Ранняя остановка прекращает обучение, когда ошибка на валидационной выборке начинает расти.

Аугментация данных искусственно расширяет обучающую выборку путём случайных преобразований: поворотов, отражений, обрезки изображений. Это особенно эффективно в задачах компьютерного зрения.

Сверточные нейронные сети

Сверточные сети специализируются на обработке данных с пространственной структурой — изображений и сигналов. Их ключевая особенность — свёрточные слои, которые применяют обучаемые фильтры к небольшым участкам входа, сканируя его целиком.

Фильтр нижних слоёв выделяет простые признаки: границы, углы, текстуры. Фильтры более глубоких слоёв комбинируют их в сложные паттерны: глаза, колёса, лица. Пулинг уменьшает пространственное разрешение, повышая инвариантность к сдвигам.

Архитектуры ResNet, VGG и EfficientNet демонстрируют, как увеличение глубины и остаточные связи позволяют обучать сети из сотен слоёв, достигая сверхчеловеческой точности в классификации изображений.

Трансформеры и механизм внимания

Механизм внимания позволяет сети взвешивать значимость различных элементов входной последовательности. Самовнимание вычисляет три представления для каждого токена: запрос, ключ и значение. Совпадение запроса с ключом определяет вес, с которым значение одного токена учитывается при формировании представления другого.

Трансформер полностью отказывается от рекуррентности и строится на самовнимании. Это обеспечивает параллелизацию вычислений и возможность моделировать дальние зависимости в тексте. Модели GPT и BERT, основанные на трансформерах, совершили революцию в обработке естественного языка.

Заключение

Нейронные сети прошли путь от теоретической модели до технологии, определяющей целые отрасли. Понимание их архитектуры, принципов обучения и методов регуляризации необходимо для эффективного применения и развития новых подходов в машинном обучении.`;

import { createContext, useContext, useEffect, useState, ReactNode } from 'react';
import { User, Session } from '@supabase/supabase-js';
import { supabase } from '../lib/supabase';

interface AuthContextType {
  user: User | null;
  session: Session | null;
  loading: boolean;
  signOut: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType>({
  user: null, session: null, loading: true,
  signOut: async () => {}
});

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
      setUser(session?.user ?? null);
      setLoading(false);
    });

    return () => subscription.unsubscribe();
  }, []);

  const signOut = async () => { await supabase.auth.signOut(); };

  return (
    <AuthContext.Provider value={{ user, session, loading, signOut }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);

import { useState } from 'react';
import { HistoryItem } from '../types';

const STORAGE_KEY = 'studyforge_history';
const MAX_ITEMS = 10;

export function useHistory() {
  const load = (): HistoryItem[] => {
    try { return JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]'); }
    catch { return []; }
  };

  const [items, setItems] = useState<HistoryItem[]>(load);

  const add = (item: HistoryItem) => {
    setItems(prev => {
      const next = [item, ...prev].slice(0, MAX_ITEMS);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      return next;
    });
  };

  const remove = (id: string) => {
    setItems(prev => {
      const next = prev.filter(i => i.id !== id);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      return next;
    });
  };

  const clear = () => {
    setItems([]);
    localStorage.removeItem(STORAGE_KEY);
  };

  return { items, add, remove, clear };
}

export async function syncFlashcards(userId: string, materialId: string, cards: any[]) {
  // Добавляет карточки только если их ещё нет
  const existing = await supabase
    .from('flashcard_progress')
    .select('question')
    .eq('material_id', materialId);
  const existingQ = new Set((existing.data || []).map((r: any) => r.question));
  const toInsert = cards
    .filter((c: any) => !existingQ.has(c.question))
    .map((c: any) => ({ user_id: userId, material_id: materialId, question: c.question, answer: c.answer }));
  if (toInsert.length > 0) {
    await supabase.from('flashcard_progress').insert(toInsert);
  }
}

export async function updateCardProgress(cardId: string, quality: number) {
  // SM-2 алгоритм
  const { data: card } = await supabase.from('flashcard_progress').select('*').eq('id', cardId).single();
  if (!card) return;

  let { ease_factor, interval_days, repetitions } = card;
  if (quality >= 3) {
    if (repetitions === 0) interval_days = 1;
    else if (repetitions === 1) interval_days = 6;
    else interval_days = Math.round(interval_days * ease_factor);
    repetitions += 1;
    ease_factor = Math.max(1.3, ease_factor + 0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02));
  } else {
    repetitions = 0;
    interval_days = 1;
  }

  const next_review = new Date();
  next_review.setDate(next_review.getDate() + interval_days);

  await supabase.from('flashcard_progress').update({
    ease_factor, interval_days, repetitions,
    next_review: next_review.toISOString()
  }).eq('id', cardId);
}

function CircularProgress({ value, size = 80, strokeWidth = 6 }: { value: number; size?: number; strokeWidth?: number }) {
  const radius = (size - strokeWidth) / 2;
  const circumference = 2 * Math.PI * radius;
  const offset = circumference - (value / 100) * circumference;
  return (
    <svg width={size} height={size} className="transform -rotate-90">
      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#e2e8f0" strokeWidth={strokeWidth} />
      <motion.circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#4F46E5" strokeWidth={strokeWidth}
        strokeLinecap="round" strokeDasharray={circumference} initial={{ strokeDashoffset: circumference }}
        animate={{ strokeDashoffset: offset }} transition={{ duration: 0.6, ease: 'easeOut' }} />
    </svg>
  );
}

function AgentTimer({ status }: { status: string }) {
  const [elapsed, setElapsed] = useState(0);
  const startRef = useRef<number | null>(null);
  const frameRef = useRef<number>(0);

  useEffect(() => {
    if (status === 'running') {
      startRef.current = Date.now();
      const tick = () => {
        if (startRef.current) setElapsed((Date.now() - startRef.current) / 1000);
        frameRef.current = requestAnimationFrame(tick);
      };
      frameRef.current = requestAnimationFrame(tick);
      return () => cancelAnimationFrame(frameRef.current);
    } else {
      cancelAnimationFrame(frameRef.current);
    }
  }, [status]);

  if (status === 'idle') return null;
  return status === 'done'
    ? <span className="text-xs text-emerald-500 tabular-nums">{elapsed.toFixed(1)}с ✓</span>
    : <span className="text-xs text-slate-400 tabular-nums">{elapsed.toFixed(1)}с</span>;
}

export default function SummaryView({ data }: { data: any }) {
  const content = data?.content || data?.raw || '';
  if (!content) return <p className="text-slate-400">Нет данных</p>;

  const sections = content.split(/(?=^## )/gm).filter(Boolean);

  const iconMap: Record<string, string> = {
    '📌': '📌', '🔑': '🔑', '📖': '📖', '📚': '📚', '💡': '💡',
  };

  return (
    <div className="space-y-1">
      {sections.map((sec: string, i: number) => {
        const match = sec.match(/^## (.+)/);
        if (!match) return (
          <div key={i} className="prose prose-slate max-w-none text-slate-700">
            <ReactMarkdown>{sec}</ReactMarkdown>
          </div>
        );
        const title = match[1].trim();
        const body = sec.replace(/^## .+\n?/, '');
        const icon = Object.keys(iconMap).find(k => title.includes(k)) || '📄';
        const cleanTitle = title.replace(/[📌🔑📖📚💡]/g, '').trim();

        return (
          <SectionBlock key={i} icon={icon} title={cleanTitle}>
            <div className="prose prose-slate prose-sm max-w-none">
              <ReactMarkdown>{body}</ReactMarkdown>
            </div>
          </SectionBlock>
        );
      })}
    </div>
  );
}

            <div className="grid grid-cols-2 gap-4 max-w-3xl w-full">
              {fsQ.options?.map((opt: string, j: number) => {
                const selected = fsAnswers[fsIdx] === opt;
                return (
                  <div key={j} onClick={() => fsSelect(opt)}

                  </div>
                );
              })}
            </div>
            {fsAnswers[fsIdx] && (

                onClick={fsNext}
                className="mt-10 px-8 py-3 bg-brand-600 hover:bg-brand-700 text-white rounded-xl text-lg font-medium transition">
                {fsIdx + 1 < fsTotal ? 'Следующий вопрос →' : 'Завершить тест'}
              </motion.button>
            )}
          </div>
        ) : (

            <div className="max-w-3xl w-full space-y-4">
              {quiz.map((q: any, i: number) => {
                const isCorrect = fsAnswers[i]?.startsWith(q.correct);
                return (
                  <div key={i} className={'rounded-xl p-5 border ' + (isCorrect ? 'bg-emerald-900/30 border-emerald-700' : 'bg-rose-900/20 border-rose-800')}>

                    </p>
                  </div>
                );
              })}
            </div>
            <button onClick={fsExit}
              className="mt-8 px-6 py-2.5 bg-white text-slate-900 rounded-lg font-medium hover:bg-slate-100 transition">
              Закрыть
            </button>
          </div>
        )}
      </div>

  }

  return (
    <div className="relative">
      {!submitted && (
        <button onClick={() => setFsOpen(true)}

          </div>

          const isWrong = submitted && selected && !isCorrect;

                    cls += 'bg-slate-50 border-slate-100 text-slate-400';
                  }

                    <div key={j} className={'flex items-center gap-2 ' + cls}
                      onClick={() => handleSelect(i, opt)}>
                      {isCorrectOpt && <CheckCircle2 className="w-4 h-4 text-emerald-500 shrink-0" />}
                      {isWrongOpt && <XCircle className="w-4 h-4 text-rose-500 shrink-0" />}
                      <span>{opt}</span>
                    </div>
                  );
                })}
              </div>
              {submitted && q.explanation && (
                <p className="mt-3 text-sm text-slate-500 italic border-t border-slate-100 pt-3">💡 {q.explanation}</p>

          <p className="text-sm text-slate-400 mb-2">
            Отвечено {Object.keys(answers).length} из {total} вопросов

              Проверить ответы

      )}
    </div>
  );
}

const validatePassword = (pwd: string) => {
  if (!pwd) return '';
  if (pwd.length < 6) return `Слишком короткий — ещё ${6 - pwd.length} символов`;
  if (pwd.length < 8) return 'Слабый пароль — добавь ещё символы';
  return '';
};

const validateName = (name: string) => {

  return '';
};

import { useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { supabase } from '../lib/supabase';

export default function AuthCallback() {
  const navigate = useNavigate();

  useEffect(() => {
    // Обрабатываем токены из URL (email confirmation, magic link)
    supabase.auth.getSession().then(({ data: { session } }) => {
      if (session) {
        navigate('/dashboard', { replace: true });
      } else {
        // Слушаем событие входа после обработки токена
        const { data: { subscription } } = supabase.auth.onAuthStateChange((event, session) => {
          if (event === 'SIGNED_IN' && session) {
            navigate('/dashboard', { replace: true });
            subscription.unsubscribe();
          } else if (event === 'TOKEN_REFRESHED') {
            navigate('/dashboard', { replace: true });
            subscription.unsubscribe();
          }
        });
        // Если через 3 секунды ничего — на логин
        setTimeout(() => {
          navigate('/login', { replace: true });
          subscription.unsubscribe();
        }, 3000);
      }
    });
  }, []);

  return (
    <div className="min-h-screen bg-slate-900 flex flex-col items-center justify-center gap-4">
      <div className="w-10 h-10 border-2 border-indigo-500 border-t-transparent rounded-full animate-spin" />
      <p className="text-slate-400 text-sm">Выполняем вход...</p>
    </div>
  );

  id: number;
  text: string;
}

export default function Analyze() {
  const [state, setState] = useState<AnalysisState>({
    phase: 'upload', agents: initAgents, results: null, settings: DEFAULT_SETTINGS,
  });
  const [isDemo, setIsDemo] = useState(false);
  const [toasts, setToasts] = useState<Toast[]>([]);
  const toastId = useRef(0);
  const analysisStartRef = useRef<number>(0);
  const sourceFileNameRef = useRef<string>('');
  const textLengthRef = useRef<number>(0);
  const { user } = useAuth();
  const navigate = useNavigate();

  const addToast = (text: string) => {
    const id = ++toastId.current;
    setToasts(prev => [...prev, { id, text }]);
    setTimeout(() => setToasts(prev => prev.filter(t => t.id !== id)), 3000);
  };

  const handleUpload = async (file: File) => {
    analysisStartRef.current = Date.now();
    sourceFileNameRef.current = file.name;
    setState({ phase: 'extracting', agents: initAgents.map(a => ({ ...a, status: 'idle' as const })), results: null, settings: state.settings, sourceFileName: file.name });
    const fd = new FormData();
    fd.append('file', file);
    fd.append('summary_detail', state.settings.summaryDetail);
    fd.append('flashcard_count', String(state.settings.flashcardCount));
    fd.append('quiz_count', String(state.settings.quizCount));
    fd.append('quiz_difficulty', state.settings.quizDifficulty);
    fd.append('quiz_type', state.settings.quizType);
    try {
      const res = await fetch('/api/analyze', { method: 'POST', body: fd });
      if (!res.ok) { const e = await res.json(); throw new Error(e.detail || 'Ошибка сервера'); }
      const reader = res.body?.getReader();
      if (!reader) throw new Error('No body');
      const dec = new TextDecoder(); let buf = '';
      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        buf += dec.decode(value, { stream: true });
        const lines = buf.split('\n'); buf = lines.pop() || '';
        for (const l of lines) {
          if (l.startsWith('data: ')) {
            try { handleSSE(JSON.parse(l.slice(6))); } catch {}
          }
        }
      }
    } catch (err: any) { setState(p => ({ ...p, phase: 'error', error: err.message })); }
  };

  const agentDoneLabels: Record<string, string> = {
    summarizer: '✅ Конспект готов',
    flashcards: '✅ Карточки готовы',
    quiz: '✅ Тест готов',
  };

  const handleSSE = (d: any) => {
    if (d.type === 'text_extracted') { 
      setState(p => ({ ...p, phase: 'analyzing', textLength: d.length })); 
      textLengthRef.current = d.length;
      (window as any)._lastExtractedText = d.text; // Сохраняем текст из события
    }
    else if (d.type === 'agent_start') setState(p => ({ ...p, agents: p.agents.map(a => a.id === d.agent ? { ...a, status: 'running' } : a) }));
    else if (d.type === 'agent_done') {
      addToast(agentDoneLabels[d.agent] || '✅ Агент завершён');
      setState(p => ({ ...p, agents: p.agents.map(a => a.id === d.agent ? { ...a, status: 'done', progress: d.progress / d.total } : a) }));
    }
    else if (d.type === 'results') {
      const results = { summarizer: d.summarizer, flashcards: d.flashcards, quiz: d.quiz };
      const dur = analysisStartRef.current
        ? Math.round((Date.now() - analysisStartRef.current) / 1000)
        : 0;
      setState(p => ({ ...p, phase: 'done', results, analysisDuration: dur || undefined }));

      if (user) {
        const title = sourceFileNameRef.current
          ? sourceFileNameRef.current.replace(/\.[^/.]+$/, '')
          : 'Без названия';

        (async () => {
          const { data: inserted, error } = await supabase
            .from('materials')
            .insert({
              user_id: user.id,
              title,
              subject: '',
              original_text: textLengthRef.current > 0 ? (window as any)._lastExtractedText || '' : '',
              summary_json: results.summarizer,
              flashcards_json: results.flashcards,
              quiz_json: results.quiz,
            })
            .select('id')
            .single();

          if (error) {
            console.error('Supabase insert error:', error);
            return;
          }

          const cards = results.flashcards?.flashcards || [];
          if (cards.length > 0 && inserted?.id) {
            await syncFlashcards(user.id, inserted.id, cards);
          }
        })();
      }
    }
    else if (d.type === 'error') setState(p => ({ ...p, phase: 'error', error: d.message }));
  };

  const reset = () => { setState({ phase: 'upload', agents: initAgents, results: null, settings: DEFAULT_SETTINGS }); setIsDemo(false); };

  const handleDemo = () => {
    setIsDemo(true);
    const file = new File([DEMO_TEXT], 'demo_lecture.txt', { type: 'text/plain' });
    handleUpload(file);
  };

  return (
    <div className="p-6 max-w-3xl mx-auto">
      <AnimatePresence mode="wait">
        {state.phase === 'upload' && (
          <motion.div key="upload" initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -20 }}>
            <UploadZone settings={state.settings} onSettingsChange={s => setState(p => ({ ...p, settings: s }))} onUpload={handleUpload} onDemo={handleDemo} />
          </motion.div>
        )}
        {(state.phase === 'extracting' || state.phase === 'analyzing') && (
          <motion.div key="analyzing" initial={{ opacity: 0 }} animate={{ opacity: 1 }}>
            <AgentCards agents={state.agents} phase={state.phase} isDemo={isDemo} />
          </motion.div>
        )}
        {state.phase === 'done' && state.results && (
          <motion.div key="results" initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
            <ResultsPanel results={state.results} onReset={reset} textLength={state.textLength} analysisDuration={state.analysisDuration} sourceFileName={state.sourceFileName} />
            <div className="mt-4 text-center">
              <button
                onClick={() => navigate('/library')}
                className="px-5 py-2 bg-indigo-600 hover:bg-indigo-700 text-white rounded-lg text-sm font-medium transition"
              >
                📚 Перейти в библиотеку
              </button>
            </div>
          </motion.div>
        )}
        {state.phase === 'error' && (
          <motion.div key="error" initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="text-center py-16">
            <p className="text-4xl mb-3">⚠️</p>
            <h2 className="text-lg font-semibold text-slate-900 mb-1">Ошибка</h2>
            <p className="text-slate-500 mb-4 text-sm">{state.error}</p>
            <button onClick={reset} className="px-5 py-2 bg-brand-600 text-white rounded-lg text-sm hover:bg-brand-700 transition">Попробовать снова</button>
          </motion.div>
        )}
      </AnimatePresence>

      {/* Toasts */}
      <div className="fixed bottom-6 right-6 z-50 flex flex-col gap-2">
        <AnimatePresence>
          {toasts.map(t => (
            <motion.div key={t.id}
              initial={{ opacity: 0, x: 40, scale: 0.95 }}
              animate={{ opacity: 1, x: 0, scale: 1 }}
              exit={{ opacity: 0, x: 40, scale: 0.95 }}
              className="bg-white border border-slate-200 shadow-lg rounded-lg px-4 py-2.5 text-sm text-slate-700 font-medium">
              {t.text}
            </motion.div>
          ))}
        </AnimatePresence>
      </div>
    </div>
  );
}

function CircularProgress({ value, size = 100, strokeWidth = 8 }: { value: number; size?: number; strokeWidth?: number }) {
  const radius = (size - strokeWidth) / 2;
  const circumference = 2 * Math.PI * radius;
  const offset = circumference - (value / 100) * circumference;
  const color = value >= 80 ? '#10b981' : value >= 50 ? '#f59e0b' : '#ef4444';
  return (
    <svg width={size} height={size} className="transform -rotate-90">
      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#e2e8f0" strokeWidth={strokeWidth} />
      <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
        strokeLinecap="round" strokeDasharray={circumference} strokeDashoffset={offset} />
    </svg>
  );
}

export default function Review() {

      if (quality >= 3) setSessionStats(p => ({ ...p, correct: p.correct + 1 }));

      else { setIdx(p => p + 1); setRevealed(false); }

          Новый анализ
        </button>
      </div>
    </div>
  );

  // Session done
  if (done) {
    const total = sessionStats.correct + sessionStats.wrong;

          </div>
        </div>
        <p className="text-slate-500 mb-1">Правильно: <span className="font-semibold text-emerald-600">{sessionStats.correct}</span> из {total}</p>

            <Zap className="w-4 h-4" />Повторить ещё

          <span>{sessionStats.correct} ✓ · {sessionStats.wrong} ✗</span>
        </div>
        <div className="h-2 bg-slate-100 rounded-full overflow-hidden">
          <div className="h-full bg-indigo-500 rounded-full transition-all" style={{ width: `${((idx) / cards.length) * 100}%` }} />
        </div>
      </div>

      {/* Card */}
      <motion.div
        key={idx}
        initial={{ opacity: 0, x: 30 }}
        animate={{ opacity: 1, x: 0 }}
        className="card min-h-48 p-8 flex flex-col items-center justify-center cursor-pointer"
        onClick={() => !revealed && setRevealed(true)}
      >

        <p className="text-xl font-medium text-slate-900 text-center mb-6">{card.question}</p>

        {revealed && (
          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="w-full border-t border-slate-100 pt-6">
            <span className="text-xs uppercase text-emerald-500 tracking-wider mb-2 block text-center">Ответ</span>
            <p className="text-slate-700 text-center">{card.answer}</p>

export default function Chat() {
  const { user } = useAuth();
  const [materials, setMaterials] = useState<any[]>([]);
  const [selectedId, setSelectedId] = useState('');
  const [messages, setMessages] = useState<{ role: string; content: string }[]>([]);
  const [input, setInput] = useState('');
  const [loading, setLoading] = useState(false);
  const [fetchingHistory, setFetchingHistory] = useState(false);
  const [isFullScreen, setIsFullScreen] = useState(false);
  const messagesEndRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!user) return;
    getMaterials(user.id).then(setMaterials).catch(console.error);
  }, [user]);

  // Загрузка истории при выборе материала
  useEffect(() => {
    if (!selectedId || !user) {
      setMessages([]);
      return;
    }

    const loadHistory = async () => {
      setFetchingHistory(true);
      const { data, error } = await supabase
        .from('chat_messages')
        .select('role, content')
        .eq('material_id', selectedId)
        .eq('user_id', user.id)
        .order('created_at', { ascending: true });
      
      if (!error && data) {
        setMessages(data);
      }
      setFetchingHistory(false);
    };

    loadHistory();
  }, [selectedId, user]);

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  const selectedMaterial = materials.find(m => m.id === selectedId);
  
  // Контекст: берем либо оригинал, либо конспект (что есть)
  const getContext = () => {
    if (!selectedMaterial) return '';
    const original = selectedMaterial.original_text || '';
    const summary = typeof selectedMaterial.summary_json === 'string' 
      ? selectedMaterial.summary_json 
      : (selectedMaterial.summary_json?.content || '');
    
    // Если есть оригинал — берем его (он полнее), иначе конспект
    return (original.length > 50 ? original : summary).slice(0, 30000);
  };

  const handleSend = async () => {
    if (!input.trim() || !selectedId || !user || loading) return;
    
    const userMsg = { role: 'user', content: input };
    setMessages(prev => [...prev, userMsg]);
    setInput('');
    setLoading(true);

    try {
      // 1. Сохраняем сообщение пользователя в БД
      await supabase.from('chat_messages').insert({
        user_id: user.id,
        material_id: selectedId,
        role: 'user',
        content: userMsg.content
      });

      // 2. Запрос к AI
      const res = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          message: userMsg.content, 
          context: getContext(), 
          history: messages.slice(-6) // Отправляем последние 6 сообщений для памяти
        }),
      });
      
      const data = await res.json();
      const assistantMsg = { role: 'assistant', content: data.reply };
      
      setMessages(prev => [...prev, assistantMsg]);

      // 3. Сохраняем ответ AI в БД
      await supabase.from('chat_messages').insert({
        user_id: user.id,
        material_id: selectedId,
        role: 'assistant',
        content: assistantMsg.content
      });

    } catch (err) {
      console.error(err);
      setMessages(prev => [...prev, { role: 'assistant', content: 'Ошибка при обращении к AI. Попробуйте снова.' }]);
    } finally {
      setLoading(false);
    }
  };

    if (!selectedId || !user || !confirm('Очистить историю чата?')) return;

          </div>

          </div>
        </div>
        
        <div className="flex items-center gap-2">

            onClick={() => setIsFullScreen(!isFullScreen)}
            className="p-2 text-slate-400 hover:text-indigo-600 transition rounded-lg hover:bg-indigo-50"
            title={isFullScreen ? "Уменьшить" : "На весь экран"}
          >
            {isFullScreen ? <Minimize2 className="w-5 h-5" /> : <Maximize2 className="w-5 h-5" />}
          </button>
          
          {messages.length > 0 && (
            <button onClick={clearChat} className="p-2 text-slate-400 hover:text-rose-500 transition rounded-lg hover:bg-rose-50" title="Очистить чат">
              <Trash2 className="w-5 h-5" />

          {materials.map(m => (
            <option key={m.id} value={m.id}>{m.title}</option>
          ))}
        </select>
      </div>

      {/* Chat area */}
      <div className="flex-1 bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden flex flex-col relative">
        {!selectedId ? (

-- 1. materials — Проанализированные материалы
CREATE TABLE materials (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         uuid NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  title           text NOT NULL DEFAULT '',
  subject         text NOT NULL DEFAULT '',
  original_text   text NOT NULL DEFAULT '',
  summary_json    jsonb,
  flashcards_json jsonb,
  quiz_json       jsonb,
  created_at      timestamptz NOT NULL DEFAULT now()
);
ALTER TABLE materials ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users can read own materials" ON materials FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "Users can insert own materials" ON materials FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users can update own materials" ON materials FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "Users can delete own materials" ON materials FOR DELETE USING (auth.uid() = user_id);
CREATE INDEX idx_materials_user_id ON materials(user_id);

-- 2. profiles — Профиль пользователя (автосоздание при регистрации)
CREATE TABLE profiles (
  id               uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name        text NOT NULL DEFAULT '',
  streak_days      int NOT NULL DEFAULT 0,
  last_active_date date,
  total_reviews    int NOT NULL DEFAULT 0,
  total_materials  int NOT NULL DEFAULT 0,
  created_at       timestamptz NOT NULL DEFAULT now(),
  updated_at       timestamptz NOT NULL DEFAULT now()
);
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users can read own profile" ON profiles FOR SELECT USING (auth.uid() = id);
CREATE POLICY "Users can update own profile" ON profiles FOR UPDATE USING (auth.uid() = id);
CREATE POLICY "Users can insert own profile" ON profiles FOR INSERT WITH CHECK (auth.uid() = id);

CREATE INDEX idx_flashcard_progress_user ON flashcard_progress(user_id);

-- Триггер: автосоздание профиля при регистрации
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger AS 
BEGIN
  INSERT INTO public.profiles (id, full_name)
  VALUES (NEW.id, COALESCE(NEW.raw_user_meta_data->>'full_name', ''));
  RETURN NEW;
END;
 LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
